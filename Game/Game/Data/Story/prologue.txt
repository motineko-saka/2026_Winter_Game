#

#博士
よく来たな。\n最初のパートナーを選んでくれ。

@choice "風タイプの子" wind "炎タイプの子" fire "水タイプの子" water

*wind
@give 1 5
@set starter = 1
@jump afterstarter

*fire
@give 4 5
@set starter = 2
@jump afterstarter

*water
@give 7 5
@set starter = 3
@jump afterstarter

*afterstarter
#博士

#

@battle 1 3

// battle_result: 1=勝ち 0=負け 2=逃げた 3=捕まえた

@if battle_result == 0 lost
@if battle_result == 3 caught
@jump escaped

*won
#博士

@jump fin

*caught
#博士

@jump fin

*lost
@heal
#博士

@jump fin

*escaped
#博士

@jump fin

*fin
#博士
では、行ってきなさい。
@end
